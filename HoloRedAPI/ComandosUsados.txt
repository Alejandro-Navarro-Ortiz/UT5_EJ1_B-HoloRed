CREATE KEYSPACE IF NOT EXISTS holored WITH replication = {'class':'SimpleStrategy', 'replication_factor':1};

CREATE TABLE IF NOT EXISTS holored.impactos (
    SectorId text, 
    Fecha date, 
    Timestamp timestamp,
    NaveAtacante text, 
    NaveObjetivo text, 
    DanoEscudos int,
    PRIMARY KEY ((SectorId, Fecha), Timestamp)
) WITH CLUSTERING ORDER BY (Timestamp DESC);
